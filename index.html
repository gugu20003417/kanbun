async function loadSheetData() { let url = document.getElementById('sheetUrlInput').value.trim(); if (!url) { alert("スプレッドシートのURLを入力してください。"); return; } // スプレッドシートIDを抽出してCORS回避可能なgviz URLに自動変換 const idMatch = url.match(/\/d\/([a-zA-Z0-9-_]+)/); if (idMatch && idMatch[1]) { const sheetId = idMatch[1]; url = `https://docs.google.com/spreadsheets/d/${sheetId}/gviz/tq?tqx=out:csv`; } const loading = document.getElementById('loading'); loading.style.display = 'flex'; try { const response = await fetch(url); if (!response.ok) throw new Error("データの取得に失敗しました。"); const csvText = await response.text(); const parsedData = parseCSV(csvText); if (parsedData.length === 0) { alert("有効な問題データが見つかりませんでした。"); return; } rawQuestions = [...parsedData]; questions = [...parsedData]; currentIndex = 0; score = 0; answeredStatus = {}; document.getElementById('scoreDisplay').textContent = "正解: 0"; renderQuestion(); alert(`${questions.length} 件のデータを取り込みました。`); } catch (err) { alert("読み込みエラーが発生しました。スプレッドシートの共有設定が「リンクを知っている全員」になっているか確認してください。"); console.error(err); } finally { loading.style.display = 'none'; } }
